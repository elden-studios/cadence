- [ ] **Step 5.1: Add the helper extension**

Create `Packages/BillableCore/Sources/BillableCore/Persistence/ModelContext+SaveOrLog.swift`:

```swift
import Foundation
import SwiftData
import OSLog

extension ModelContext {
    /// Save the context, swallowing the throw but surfacing the error to the
    /// system log so production issues can be diagnosed from device logs.
    ///
    /// Use this in place of `try? modelContext.save()` everywhere a silent
    /// failure would mislead the user into thinking their edit persisted.
    /// A future improvement is to plumb a UI-visible toast through this
    /// helper — for now, OSLog is the floor.
    @MainActor
    public func saveOrLog(
        _ context: String = "",
        file: String = #fileID,
        line: Int = #line
    ) {
        do {
            try self.save()
        } catch {
            let logger = Logger(subsystem: "com.eldenstudios.billable", category: "persistence")
            logger.error(
                "ModelContext save failed at \(file, privacy: .public):\(line, privacy: .public) (\(context, privacy: .public)): \(error.localizedDescription, privacy: .public)"
            )
        }
    }
}
```

```swift
import Testing
import Foundation
import SwiftData
@testable import BillableCore

@Suite("ModelContext.saveOrLog")
@MainActor
struct ModelContextSaveOrLogTests {

    @Test("saveOrLog persists changes on success")
    func savePersists() throws {
        let container = try BillableModelContainer.inMemory()
        let context = ModelContext(container)
        let client = Client(name: "Acme")
        context.insert(client)
        context.saveOrLog("test")
        // Re-fetch in a fresh context to confirm persistence
        let fetched = try context.fetch(FetchDescriptor<Client>())
        #expect(fetched.count == 1)
        #expect(fetched.first?.name == "Acme")
    }

    @Test("saveOrLog does not throw, even on no-op")
    func saveOrLogNoOpDoesNotThrow() throws {
        let container = try BillableModelContainer.inMemory()
        let context = ModelContext(container)
        // No changes pending — should be a clean no-op
        context.saveOrLog("no-op")  // does not throw
    }
}
```
