- [ ] **Step 1.4: Add a 5-second loading timeout + error surface in `SubscriptionManager`**

Right now (probably) `Product.products(for:)` returns an empty array silently on network failure, leaving `yearly` and `monthly` as nil and the UI in infinite-spinner state. Add an error/empty surface:

Open `Packages/BillableCore/Sources/BillableCore/Subscriptions/SubscriptionManager.swift`. Find the function that calls `Product.products(for:)`. Around it, add:

```swift
public enum LoadState: Sendable, Equatable {
    case idle
    case loading
    case ready
    case failed(String)
}

public private(set) var loadState: LoadState = .idle
```

And in the loading function, wrap the call:

```swift
public func reloadProducts() async {
    loadState = .loading
    do {
        let products = try await withTimeout(seconds: 5) {
            try await Product.products(for: Set(Self.productIDs))
        }
        if products.isEmpty {
            loadState = .failed("Pricing unavailable. Pull to retry.")
            return
        }
        // ... existing assign-to-yearly/monthly logic ...
        loadState = .ready
    } catch {
        loadState = .failed(error.localizedDescription)
    }
}

private func withTimeout<T: Sendable>(
    seconds: TimeInterval,
    operation: @escaping @Sendable () async throws -> T
) async throws -> T {
    try await withThrowingTaskGroup(of: T.self) { group in
        group.addTask { try await operation() }
        group.addTask {
            try await Task.sleep(for: .seconds(seconds))
            throw CancellationError()
        }
        let result = try await group.next()!
        group.cancelAll()
        return result
    }
}
```

(If `productIDs` doesn't exist yet as a static, declare it from the IDs you observed in the `.storekit` file.)

extension BusinessProfile {
